\noindent
เป้าหมายของงานนี้คือการฝึกสอนโมเดล (Train model) ด้วยวิธีการ \textbf{Gradient Descent (GD)} เพื่อหาพารามิเตอร์ที่เหมาะสมให้กับชุดข้อมูล $n = 100$ จุด โดยกำหนดให้โมเดลเป็นสมการพหุนามกำลังสองที่ไม่มีพจน์เชิงเส้น:

\[
  \hat{y} = C_0 + C_1 x^2
\]

% ══════════════════════════════════════════════════════════════════════════════
\needspace{8\baselineskip}
\section*{1. ฟังก์ชันความสูญเสีย (Loss Function)}

สำหรับโมเดล $\hat{y}_i = C_0 + C_1 x_i^2$ เราใช้ฟังก์ชันความสูญเสียแบบ \textbf{Mean Squared Error (MSE)}:

\[
  L(C_0, C_1) = \frac{1}{n} \sum_{i=1}^{n} \left( y_i - \hat{y}_i \right)^2
\]

เมื่อแทนค่า $\hat{y}_i$ ลงไป จะได้:

\begin{ansbox}
\[
  L(C_0, C_1) = \frac{1}{n} \sum_{i=1}^{n} \left( y_i - C_0 - C_1 x_i^2 \right)^2
\]
\end{ansbox}

% ══════════════════════════════════════════════════════════════════════════════
\needspace{10\baselineskip}
\section*{2. การหาอนุพันธ์ย่อย (Gradient of Each Coefficient)}

เพื่อใช้ Gradient Descent เราต้องหาอนุพันธ์ย่อย (Partial derivatives) ของ $L$ เทียบกับ $C_0$ และ $C_1$

ให้ $e_i = y_i - C_0 - C_1 x_i^2$ 

\textbf{Gradient เทียบกับ $C_0$:}
\begin{align*}
  \frac{\partial L}{\partial C_0} &= \frac{1}{n} \sum_{i=1}^{n} 2 e_i \cdot \frac{\partial e_i}{\partial C_0} = \frac{1}{n} \sum_{i=1}^{n} 2 e_i \cdot (-1)
\end{align*}
\begin{ansbox}
\[
  \frac{\partial L}{\partial C_0} = -\frac{2}{n} \sum_{i=1}^{n} \left( y_i - C_0 - C_1 x_i^2 \right)
\]
\end{ansbox}

\textbf{Gradient เทียบกับ $C_1$:}
\begin{align*}
  \frac{\partial L}{\partial C_1} &= \frac{1}{n} \sum_{i=1}^{n} 2 e_i \cdot \frac{\partial e_i}{\partial C_1} = \frac{1}{n} \sum_{i=1}^{n} 2 e_i \cdot (-x_i^2)
\end{align*}
\begin{ansbox}
\[
  \frac{\partial L}{\partial C_1} = -\frac{2}{n} \sum_{i=1}^{n} \left( y_i - C_0 - C_1 x_i^2 \right) x_i^2
\]
\end{ansbox}

% ══════════════════════════════════════════════════════════════════════════════
\needspace{10\baselineskip}
\section*{3. กฎการอัปเดตพารามิเตอร์ (Update Rules)}

ในแต่ละรอบการวนซ้ำ (Iteration) $t$ พารามิเตอร์จะถูกปรับค่าตรงข้ามกับ Gradient โดยมีอัตราการเรียนรู้ (Learning Rate) $\eta$ ควบคุมขนาดของก้าว:

\begin{ansbox}
\begin{align*}
  C_0^{(t+1)} &= C_0^{(t)} - \eta \cdot \frac{\partial L}{\partial C_0} \\
  C_1^{(t+1)} &= C_1^{(t)} - \eta \cdot \frac{\partial L}{\partial C_1}
\end{align*}
\end{ansbox}

% ══════════════════════════════════════════════════════════════════════════════
\needspace{8\baselineskip}
